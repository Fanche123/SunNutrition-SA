    const dueDate = new Date(`${startDate}T12:00:00`);

    const secondDue = new Date(dueDate);

      secondDueDate: secondDue.toISOString().slice(0, 10),
      capital,
      financialInterest,
      lateInterest,
      firstAmount,

      paid: index < paidCount
    };
